Manifiestos Cloud-Native (Docker & Kubernetes)

Especificación multi-etapa Dockerfile y Deployment YAML

# Etapa 1: Compilación de la aplicación Spring Boot con Maven
FROM maven:3.9-eclipse-temurin-21-alpine AS builder
WORKDIR /workspace
COPY pom.xml .
RUN mvn dependency:go-offline -B
COPY src src
RUN mvn clean package -DskipTests

# Etapa 2: Imagen de ejecución ligera con Alpine Linux
FROM eclipse-temurin:21-jre-alpine
LABEL author="Christian Camilo Sandoval Vega"
RUN addgroup -S appgroup && adduser -S appuser -G appgroup
WORKDIR /app
COPY --from=builder /workspace/target/*.jar app.jar
USER appuser:appgroup
EXPOSE 8080
ENV JAVA_OPTS="-XX:+UseG1GC -XX:MaxRAMPercentage=75.0"
ENTRYPOINT ["sh", "-c", "java $JAVA_OPTS -jar app.jar"]
apiVersion: apps/v1
kind: Deployment
metadata:
  name: metrics-service
  namespace: production-devagile
spec:
  replicas: 4
  selector:
    matchLabels:
      app: metrics-service
  template:
    metadata:
      labels:
        app: metrics-service
    spec:
      containers:
      - name: metrics-service
        image: sandoval/metrics-service:v3.1
        ports:
        - containerPort: 8082
        resources:
          requests:
            cpu: "250m"
            memory: "512Mi"
          limits:
            cpu: "1000m"
            memory: "1024Mi"
        livenessProbe:
          httpGet:
            path: /actuator/health/liveness
            port: 8082
          initialDelaySeconds: 20
